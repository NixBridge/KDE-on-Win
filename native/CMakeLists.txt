cmake_minimum_required(VERSION 3.21)
project(PlasmaWindowsNative LANGUAGES C CXX)
install(FILES "${CMAKE_CURRENT_LIST_DIR}/qt.conf" DESTINATION bin)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core DBus)
set(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../upstream/plasma-windows/src")
add_executable(trayhost
    "${SOURCE_ROOT}/trayhost/trayhost_main.cpp"
    "${SOURCE_ROOT}/trayhost/windowstrayhost.cpp"
    "${SOURCE_ROOT}/trayhost/windowstrayhost.h"
    "${SOURCE_ROOT}/trayhost/snibridge.cpp"
    "${SOURCE_ROOT}/trayhost/snibridge.h")
target_compile_definitions(trayhost PRIVATE UNICODE _UNICODE WIN32 WIN64 NOMINMAX)
target_compile_options(trayhost PRIVATE /utf-8 /permissive- /Zc:__cplusplus)
target_link_libraries(trayhost PRIVATE Qt6::Core Qt6::DBus user32 shell32 gdi32)
install(TARGETS trayhost RUNTIME DESTINATION bin)
add_executable(titlebar "${SOURCE_ROOT}/titlebar/titlebar.c")
set_property(TARGET titlebar PROPERTY AUTOMOC OFF)
target_compile_definitions(titlebar PRIVATE UNICODE _UNICODE)
target_compile_options(titlebar PRIVATE /utf-8)
target_link_libraries(titlebar PRIVATE user32 gdi32 advapi32 dwmapi)
add_executable(shellswitch WIN32 "${SOURCE_ROOT}/shellswitch/shellswitch.c")
set_property(TARGET shellswitch PROPERTY AUTOMOC OFF)
target_compile_definitions(shellswitch PRIVATE UNICODE _UNICODE)
target_compile_options(shellswitch PRIVATE /utf-8)
target_link_libraries(shellswitch PRIVATE user32 shell32 advapi32)
install(TARGETS titlebar shellswitch RUNTIME DESTINATION bin)
option(BUILD_VM_PROBES "Build the window backend acceptance probe for VM testing" OFF)
if(BUILD_VM_PROBES)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_executable(desktop-launch-probe "${CMAKE_CURRENT_LIST_DIR}/desktop-launch-probe.cpp")
    target_link_libraries(desktop-launch-probe PRIVATE Qt6::Widgets Qt6::Qml ole32)
    install(TARGETS desktop-launch-probe RUNTIME DESTINATION bin)
    find_package(Qt6 REQUIRED COMPONENTS Quick)
    add_executable(native-file-icon-probe "${CMAKE_CURRENT_LIST_DIR}/native-file-icon-probe.cpp" "${SOURCE_ROOT}/startmenu/startmenuimageprovider.cpp")
    target_link_libraries(native-file-icon-probe PRIVATE Qt6::Quick KF6::IconThemes shell32 ole32 user32 gdi32)
    install(TARGETS native-file-icon-probe RUNTIME DESTINATION bin)
    find_package(KF6IconThemes REQUIRED)
    add_executable(icon-probe "${CMAKE_CURRENT_LIST_DIR}/icon-probe.cpp")
    target_link_libraries(icon-probe PRIVATE Qt6::Gui KF6::IconThemes)
    install(TARGETS icon-probe RUNTIME DESTINATION bin)
    find_package(KF6KIO REQUIRED)
    add_executable(desktop-protocol-probe "${CMAKE_CURRENT_LIST_DIR}/desktop-protocol-probe.cpp")
    target_link_libraries(desktop-protocol-probe PRIVATE KF6::KIOCore)
    install(TARGETS desktop-protocol-probe RUNTIME DESTINATION bin)
    find_package(Qt6 REQUIRED COMPONENTS Gui Qml)
    add_executable(qml-import-probe "${CMAKE_CURRENT_LIST_DIR}/qml-import-probe.cpp")
    target_link_libraries(qml-import-probe PRIVATE Qt6::Gui Qt6::Qml)
    install(TARGETS qml-import-probe RUNTIME DESTINATION bin)
    find_package(KF6WindowSystem 6.26 REQUIRED)
    add_executable(kwindowsystem-probe "${CMAKE_CURRENT_LIST_DIR}/../upstream/plasma-windows/probe/kwindowsystem-probe/main.cpp")
    target_compile_options(kwindowsystem-probe PRIVATE /utf-8)
    target_link_libraries(kwindowsystem-probe PRIVATE Qt6::Gui KF6::WindowSystem user32)
    install(TARGETS kwindowsystem-probe RUNTIME DESTINATION bin)
endif()
option(BUILD_PLASMA_MODULES "Build the Windows launch menu and native applets" OFF)
if(BUILD_PLASMA_MODULES)
    install(FILES "${CMAKE_CURRENT_LIST_DIR}/default-panel.js"
        DESTINATION bin/data/plasma/layout-templates/org.kde.plasma.desktop.defaultPanel/contents
        RENAME layout.js)
    find_package(Qt6 REQUIRED COMPONENTS Gui Qml Quick Network)
    find_package(KF6CoreAddons REQUIRED)
    find_package(KF6Config REQUIRED)
    find_package(KF6I18n REQUIRED)
    find_package(Plasma REQUIRED)
    find_package(PlasmaQuick REQUIRED)
    find_package(LibKWorkspace REQUIRED)
    add_library(startmenuplugin MODULE
        "${SOURCE_ROOT}/startmenu/plugin.cpp"
        "${SOURCE_ROOT}/startmenu/startmenumodel.cpp"
        "${SOURCE_ROOT}/startmenu/startmenumodel.h"
        "${SOURCE_ROOT}/startmenu/startmenuimageprovider.cpp"
        "${SOURCE_ROOT}/startmenu/poweractions.cpp"
        "${SOURCE_ROOT}/startmenu/poweractions.h")
    set_target_properties(startmenuplugin PROPERTIES PREFIX "")
    target_compile_definitions(startmenuplugin PRIVATE UNICODE _UNICODE NOMINMAX)
    target_compile_options(startmenuplugin PRIVATE /utf-8 /Zc:__cplusplus)
    target_link_libraries(startmenuplugin PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick KF6::IconThemes PW::KWorkspace ole32 user32 shell32 gdi32)
    install(TARGETS startmenuplugin RUNTIME DESTINATION qml/org/kde/plasma/windowsmenu LIBRARY DESTINATION qml/org/kde/plasma/windowsmenu)
    install(FILES "${SOURCE_ROOT}/startmenu/qmldir" DESTINATION qml/org/kde/plasma/windowsmenu)
    foreach(name volumewin imewin windowsmenu)
        set(sources "${SOURCE_ROOT}/applets/${name}/${name}.cpp")
        if(name STREQUAL volumewin)
            list(APPEND sources "${SOURCE_ROOT}/applets/${name}/volumecontroller.cpp" "${SOURCE_ROOT}/applets/${name}/volumecontroller.h")
        elseif(name STREQUAL imewin)
            list(APPEND sources "${SOURCE_ROOT}/applets/${name}/imecontroller.cpp" "${SOURCE_ROOT}/applets/${name}/imecontroller.h")
        endif()
        add_library(${name} MODULE ${sources})
        set_target_properties(${name} PROPERTIES PREFIX "" OUTPUT_NAME "org.kde.plasma.${name}")
        target_compile_definitions(${name} PRIVATE UNICODE _UNICODE NOMINMAX)
        target_compile_options(${name} PRIVATE /utf-8 /Zc:__cplusplus)
        target_link_libraries(${name} PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network Plasma::Plasma Plasma::PlasmaQuick KF6::CoreAddons KF6::I18n ole32 uuid user32)
        install(TARGETS ${name} RUNTIME DESTINATION plugins/plasma/applets LIBRARY DESTINATION plugins/plasma/applets)
        install(DIRECTORY "${SOURCE_ROOT}/applets/${name}/plasmoid/" DESTINATION "bin/data/plasma/plasmoids/org.kde.plasma.${name}")
    endforeach()
endif()
