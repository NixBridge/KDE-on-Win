cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(KDEWindowsIntegration LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
add_library(kde-controls SHARED controls.cpp controls.def)
add_executable(kde-style-host WIN32 style-host.cpp)
foreach(target kde-controls kde-style-host)
    set_target_properties(${target} PROPERTIES AUTOMOC OFF MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX)
    target_compile_options(${target} PRIVATE /utf-8)
    target_link_libraries(${target} PRIVATE user32 gdi32 comctl32)
endforeach()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_executable(kde-session-host WIN32 session-host.cpp)
    add_library(kde-logon-provider SHARED logon.cpp logon.def)
    add_executable(kde-logon-probe logon-probe.cpp)
    foreach(target kde-logon-provider kde-logon-probe kde-session-host)
        set_target_properties(${target} PROPERTIES AUTOMOC OFF MSVC_RUNTIME_LIBRARY MultiThreaded)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX)
        target_compile_options(${target} PRIVATE /utf-8)
        target_link_libraries(${target} PRIVATE user32 gdi32 ole32 uuid shlwapi credui secur32 shell32)
    endforeach()
    install(TARGETS kde-logon-provider kde-logon-probe kde-session-host RUNTIME DESTINATION bin)
    install(TARGETS kde-controls kde-style-host RUNTIME DESTINATION bin)
else()
    install(TARGETS kde-controls kde-style-host RUNTIME DESTINATION bin/x86)
endif()
